    async function sha256(message) {
        const msgBuffer = new TextEncoder().encode(message);
        const hashBuffer = await crypto.subtle.digest('SHA-256', msgBuffer);
        const hashArray = Array.from(new Uint8Array(hashBuffer));
        return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
    }

    function setupRealtimeListener() {
        const activeUrl = appState.firebaseUrl;
        if (!activeUrl || !activeUrl.startsWith('http')) return;

        if (appState.eventSource) {
            appState.eventSource.close();
        }

        try {
            appState.eventSource = new EventSource(`${activeUrl}/documents.json`);
            appState.eventSource.addEventListener('put', (event) => {
                try {
                    const parsed = JSON.parse(event.data);
                    if (parsed && parsed.data) {
                        const sanitized = sanitizeDocumentList(parsed.data);
                        if (sanitized.length > 0) {
                            appState.documents = sanitized;
                            localStorage.setItem('school_cached_docs', JSON.stringify(appState.documents));
                            updateHeaderStatus();
                            if (appState.isAdminLoggedIn) renderAdminDocs();
                        }
                    }
                } catch(e){}
            });
            appState.eventSource.onerror = () => {};
        } catch(e) {
            console.warn("SSE Listener init:", e);
        }
    }

    async function syncDocsToCloud(showToastMessage = true) {

            try {
                const res = await fetch(`${activeUrl}/documents.json`, {
                    method: 'PUT',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(appState.documents)
                });

            }
        }

    async function handleAdminFileUpload(event) {
        const file = event.target.files[0];
        if (!file) return;

        if (file.type !== 'application/pdf') {
            showToast('لطفاً تنها فایل با فرمت PDF انتخاب فرمایید.', 'error');
            return;
        }

        const spinner = document.getElementById('uploadSpinner');
        spinner.classList.remove('hidden');

        try {
            const buffer = await file.arrayBuffer();
            const loadingTask = pdfjsLib.getDocument({ data: buffer });
            const pdf = await loadingTask.promise;

            let extractedText = '';
            for (let pageNum = 1; pageNum <= pdf.numPages; pageNum++) {
                const page = await pdf.getPage(pageNum);
                const textContent = await page.getTextContent();
                const pageText = textContent.items.map(item => item.str).join(' ');
                extractedText += `\n[صفحه ${pageNum}]:\n${pageText}\n`;
            }

            if (!extractedText.trim()) {
                showToast('متنی در این فایل یافت نشد. ممکن است اسکن تصویری باشد.', 'error');
                return;
            }

            const newDoc = {
                id: 'doc-' + Date.now(),
                title: file.name.replace('.pdf', ''),
                date: new Date().toLocaleDateString('fa-IR'),
                content: extractedText.trim()
            };

            appState.documents.push(newDoc);
            await syncDocsToCloud(true);
            showToast(`فایل «${file.name}» در پایگاه سراسری ذخیره شد!`);
            event.target.value = '';
        } catch (e) {
            console.error("PDF Read Error:", e);
            showToast('خطا در پردازش فایل PDF.', 'error');
        } finally {
            spinner.classList.add('hidden');
        }
    }

        const newPass = document.getElementById('newPassInput').value.trim();
        const confirmPass = document.getElementById('confirmNewPassInput').value.trim();

        const hashedCurrent = await sha256(currentPass);
        if (hashedCurrent !== appState.adminPassHash) {
            showToast('رمز عبور فعلی نادرست است!', 'error');
            return;
        }

        if (!newPass || newPass.length < 5) {
            showToast('رمز جدید باید حداقل ۵ رقم/حرف باشد.', 'error');

        }

            showToast('تکرار رمز با رمز جدید یکسان نیست!', 'error');
            return;
        }

        const hashedNew = await sha256(newPass);

        localStorage.setItem('school_admin_pass_hash', hashedNew);
        closeChangePasswordModal();
        showToast('رمز عبور مدیریت با موفقیت تغییر یافت.');
    }

    // موتور جستجوگر و تطبیق متن هوشمند بدون نیاز به API (آفلاین / Fail-Safe)
    function findAnswerInDocuments(question) {
        if (!appState.documents || appState.documents.length === 0) {
            return "ولی گرامی؛ در حال حاضر بخشنامه‌ای در سامانه ثبت نشده است. لطفاً با دفتر آموزشگاه تماس حاصل فرمایید.";
        }

        const stopWords = new Set(['از', 'به', 'در', 'با', 'که', 'این', 'آن', 'برای', 'چیست', 'کیست', 'چگونه', 'است', 'آیا', 'و', 'یا', 'را', 'ساعت', 'های']);
        const tokens = question.toLowerCase().replace(/[\?\؟\,\،\.\:]/g, ' ').split(/\s+/).filter(w => w.length > 1 && !stopWords.has(w));

        let results = [];
        appState.documents.forEach(doc => {
            const sections = doc.content.split(/\n{2,}|\r\n{2,}|\n(?=\[|---)/);
            sections.forEach(sec => {
                let score = 0;
                const lowerSec = sec.toLowerCase();
                tokens.forEach(tok => {
                    if (lowerSec.includes(tok)) score += 5;
                });
                if (score > 0) {
                    results.push({ title: doc.title, text: sec.trim(), score: score });
                }
            });
        });

        results.sort((a, b) => b.score - a.score);

        if (results.length === 0) {
            return `ولی ارجمند؛ 🌸\nدرباره موضوع مطرح‌شده مصوبه‌ای در اسناد رسمی مدرسه ثبت نشده است.\nجهت راهنمایی دقیق‌تر، لطفاً در ساعات اداری با دفتر آموزشگاه تماس حاصل فرمایید.`;
        }

        let output = `ولی گرامی و ارجمند؛ بر مبنای آخرین مصوبات آموزشگاه عارفان توس:\n\n`;
        results.slice(0, 2).forEach(r => {
            output += `📌 **${r.title}**:\n${r.text}\n\n`;
        });
        output += `💡 *پاسخ استخراج‌شده از پایگاه مصوبات مدرسه*`;
        return output;
    }

        const container = document.getElementById('messagesContainer');
        const wrapper = document.createElement('div');
        const isUser = role === 'user';
        wrapper.className = `flex items-start gap-2.5 ${isUser ? 'justify-end' : ''}`;

        wrapper.innerHTML = `
            ${!isUser ? `
                <div class="w-9 h-9 rounded-2xl bg-gradient-to-tr from-rosebrand-500 to-violetbrand-600 text-white flex items-center justify-center shrink-0 shadow-btn-3d text-sm">
                    <i class="fa-solid fa-comments"></i>
                </div>
            ` : ''}
            <div class="max-w-[85%] ${isUser ? 'bubble-user bg-gradient-to-l from-rosebrand-600 to-rose-600 text-white shadow-btn-3d' : 'bubble-ai bg-white text-slate-800 border border-slate-100 shadow-card-3d'} p-4 text-xs sm:text-sm leading-relaxed">
                ${!isUser ? '<p class="font-extrabold text-rosebrand-600 text-[11px] mb-1 flex items-center gap-1.5"><i class="fa-solid fa-sparkles text-amber-500"></i> دستیار هوشمند عارفان توس</p>' : ''}
                <div class="font-medium whitespace-pre-wrap">${text}</div>
            </div>
            ${isUser ? `
                <div class="w-9 h-9 rounded-2xl bg-slate-200 text-slate-700 flex items-center justify-center shrink-0 shadow-sm text-sm border border-slate-300">
                    <i class="fa-solid fa-user"></i>
                </div>
            ` : ''}
        `;

        container.appendChild(wrapper);
        container.scrollTop = container.scrollHeight;
    }

    async function handleChatSubmit(e) {
        if (e) e.preventDefault();
        if (appState.isThinking) return;

        const input = document.getElementById('userInput');
        const text = input.value.trim();
        if (!text) return;

        input.value = '';
        input.style.height = 'auto';

        appendMessage('user', text);
        
        const typing = document.getElementById('typingIndicator');
        const container = document.getElementById('messagesContainer');
        typing.classList.remove('hidden');
        container.scrollTop = container.scrollHeight;
        
        appState.isThinking = true;
        document.getElementById('sendBtn').disabled = true;

        try {
            let answer;
            try {
                const systemPrompt = buildSchoolSystemPrompt();
                answer = await callGeminiApi(systemPrompt, text);
            } catch (apiError) {
                answer = findAnswerInDocuments(text);
            }

            typing.classList.add('hidden');
            appendMessage('model', answer);
        } catch (err) {
            typing.classList.add('hidden');
            appendMessage('model', 'خطایی در آماده‌سازی پاسخ رخ داد. لطفاً با دفتر آموزشگاه تماس حاصل فرمایید.');
        } finally {
            appState.isThinking = false;
            document.getElementById('sendBtn').disabled = false;
            input.focus();
        }
    }

    function askQuickPrompt(q) {
        document.getElementById('userInput').value = q;
        handleChatSubmit(null);
    }

    function clearChatHistory() {
        const container = document.getElementById('messagesContainer');
        while (container.children.length > 2) {
            container.removeChild(container.lastChild);
        }
        showToast('گفتگو پاکسازی شد.');
    }

    // بهینه‌سازی Three.js و جلوگیری از مصرف باتری با خاموش شدن در تب پس‌زمینه
    function initThreeJsBackground() {
        const container = document.getElementById('threejs-canvas-container');
        if (!container || typeof THREE === 'undefined') return;

        const scene = new THREE.Scene();
        const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
        camera.position.z = 30;

        const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: false, powerPreference: "low-power" });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 1.5));
        container.appendChild(renderer.domElement);

        const group = new THREE.Group();
        scene.add(group);

        const geometries = [
            new THREE.IcosahedronGeometry(1.4, 0),
            new THREE.TorusGeometry(1.5, 0.4, 12, 36),
            new THREE.SphereGeometry(1.2, 16, 16)
        ];

        const materials = [
            new THREE.MeshStandardMaterial({ color: 0xf43f5e, roughness: 0.3, wireframe: true }),
            new THREE.MeshStandardMaterial({ color: 0x8b5cf6, roughness: 0.2 }),
            new THREE.MeshStandardMaterial({ color: 0xfb7185, roughness: 0.4 })
        ];

        const meshes = [];
        for (let i = 0; i < 12; i++) {
            const geom = geometries[i % geometries.length];
            const mat = materials[i % materials.length];
            const mesh = new THREE.Mesh(geom, mat);
            
            mesh.position.x = (Math.random() - 0.5) * 40;
            mesh.position.y = (Math.random() - 0.5) * 30;
            mesh.position.z = (Math.random() - 0.5) * 20;

            const scale = Math.random() * 0.7 + 0.5;
            mesh.scale.set(scale, scale, scale);

            mesh.userData = {
                rotSpeedX: (Math.random() - 0.5) * 0.01,
                rotSpeedY: (Math.random() - 0.5) * 0.01,
                floatSpeed: Math.random() * 0.01 + 0.005,
                originalY: mesh.position.y
            };

            group.add(mesh);
            meshes.push(mesh);
        }

        const ambientLight = new THREE.AmbientLight(0xffffff, 0.9);
        scene.add(ambientLight);

        const dirLight = new THREE.DirectionalLight(0xffffff, 1);
        dirLight.position.set(20, 20, 20);
        scene.add(dirLight);

        let mouseX = 0, mouseY = 0;
        window.addEventListener('mousemove', (e) => {
            mouseX = (e.clientX / window.innerWidth - 0.5) * 2;
            mouseY = -(e.clientY / window.innerHeight - 0.5) * 2;
        });

        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });

        // مدیریت خاموش کردن انیمیشن هنگام سوئیچ کاربر به تب دیگر جهت مصرف بهینه انرژی
        document.addEventListener('visibilitychange', () => {
            appState.threeJsRunning = !document.hidden;
        });

        let clock = new THREE.Clock();
        function animate() {
            requestAnimationFrame(animate);
            if (!appState.threeJsRunning) return;

            const elapsed = clock.getElapsedTime();
            meshes.forEach(m => {
                m.rotation.x += m.userData.rotSpeedX;
                m.rotation.y += m.userData.rotSpeedY;
                m.position.y = m.userData.originalY + Math.sin(elapsed * 1.5 + m.position.x) * 0.8;
            });

            camera.position.x += (mouseX * 3 - camera.position.x) * 0.03;
            camera.position.y += (mouseY * 3 - camera.position.y) * 0.03;
            camera.lookAt(scene.position);

            renderer.render(scene, camera);
        }

        animate();
    }

    };
